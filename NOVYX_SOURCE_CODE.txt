require("dotenv").config();

const express = require("express");
const session = require("express-session");
const SQLiteStore = require("connect-sqlite3")(session);
const Database = require("better-sqlite3");
const bcrypt = require("bcryptjs");
const helmet = require("helmet");
const rateLimit = require("express-rate-limit");
const path = require("path");
const fs = require("fs");

const app = express();
const PORT = Number(process.env.PORT || 3000);
const isProduction = process.env.NODE_ENV === "production";
const dataDir = path.join(__dirname, "data");
fs.mkdirSync(dataDir, { recursive: true });

if (!process.env.SESSION_SECRET || process.env.SESSION_SECRET.length < 32) {
  console.error("SESSION_SECRET must be set to a random value of at least 32 characters.");
  process.exit(1);
}
if (!process.env.ADMIN_USERNAME || !process.env.ADMIN_PASSWORD || process.env.ADMIN_PASSWORD.length < 12) {
  console.error("Set ADMIN_USERNAME and ADMIN_PASSWORD (at least 12 characters) in .env before starting.");
  process.exit(1);
}

const db = new Database(path.join(dataDir, "novyx.sqlite"));
db.pragma("journal_mode = WAL");
db.exec(`
  CREATE TABLE IF NOT EXISTS admins (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    username TEXT NOT NULL UNIQUE,
    password_hash TEXT NOT NULL,
    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
  );
  CREATE TABLE IF NOT EXISTS requests (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    email TEXT NOT NULL,
    phone TEXT NOT NULL,
    service TEXT NOT NULL,
    budget TEXT DEFAULT '',
    details TEXT NOT NULL,
    status TEXT NOT NULL DEFAULT 'new',
    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
  );
`);

const adminExists = db.prepare("SELECT id FROM admins WHERE username = ?").get(process.env.ADMIN_USERNAME);
if (!adminExists) {
  const hash = bcrypt.hashSync(process.env.ADMIN_PASSWORD, 12);
  db.prepare("INSERT INTO admins (username, password_hash) VALUES (?, ?)").run(process.env.ADMIN_USERNAME, hash);
  console.log(`Created admin account: ${process.env.ADMIN_USERNAME}`);
}

app.disable("x-powered-by");
app.use(helmet({
  contentSecurityPolicy: {
    directives: {
      defaultSrc: ["'self'"],
      styleSrc: ["'self'", "'unsafe-inline'", "https://fonts.googleapis.com"],
      fontSrc: ["'self'", "https://fonts.gstatic.com", "data:"],
      scriptSrc: ["'self'"],
      imgSrc: ["'self'", "data:"],
      connectSrc: ["'self'"],
      upgradeInsecureRequests: isProduction ? [] : null
    }
  }
}));
app.use(express.json({ limit: "20kb" }));
app.use(express.urlencoded({ extended: false, limit: "20kb" }));
app.use(session({
  store: new SQLiteStore({ db: "sessions.sqlite", dir: dataDir }),
  secret: process.env.SESSION_SECRET,
  resave: false,
  saveUninitialized: false,
  cookie: {
    httpOnly: true,
    sameSite: "strict",
    secure: isProduction,
    maxAge: 1000 * 60 * 60 * 8
  }
}));

const loginLimiter = rateLimit({ windowMs: 15 * 60 * 1000, limit: 10, standardHeaders: "draft-7", legacyHeaders: false });
const requestLimiter = rateLimit({ windowMs: 60 * 60 * 1000, limit: 8, standardHeaders: "draft-7", legacyHeaders: false });

function requireAdmin(req, res, next) {
  if (req.session && req.session.adminId) return next();
  return res.status(401).json({ error: "يلزم تسجيل دخول الأدمن." });
}
function clean(value, max = 2000) {
  return String(value || "").trim().slice(0, max);
}

app.post("/api/requests", requestLimiter, (req, res) => {
  const name = clean(req.body.name, 100);
  const email = clean(req.body.email, 160).toLowerCase();
  const phone = clean(req.body.phone, 40);
  const service = clean(req.body.service, 100);
  const budget = clean(req.body.budget, 100);
  const details = clean(req.body.details, 3000);

  if (!name || !email || !phone || !service || !details) {
    return res.status(400).json({ error: "يرجى ملء جميع الحقول المطلوبة." });
  }
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
    return res.status(400).json({ error: "يرجى إدخال بريد إلكتروني صحيح." });
  }
  if (details.length < 15) {
    return res.status(400).json({ error: "اكتب تفاصيل أكثر عن المشروع (15 حرفًا على الأقل)." });
  }

  const result = db.prepare(`
    INSERT INTO requests (name, email, phone, service, budget, details)
    VALUES (?, ?, ?, ?, ?, ?)
  `).run(name, email, phone, service, budget, details);

  res.status(201).json({ message: "تم استلام طلبك بنجاح. سنتواصل معك لمناقشة التفاصيل.", id: result.lastInsertRowid });
});

app.post("/api/admin/login", loginLimiter, (req, res) => {
  const username = clean(req.body.username, 80);
  const password = String(req.body.password || "");
  const admin = db.prepare("SELECT * FROM admins WHERE username = ?").get(username);
  if (!admin || !bcrypt.compareSync(password, admin.password_hash)) {
    return res.status(401).json({ error: "اسم المستخدم أو كلمة المرور غير صحيحة." });
  }
  req.session.regenerate(err => {
    if (err) return res.status(500).json({ error: "تعذر إنشاء جلسة آمنة." });
    req.session.adminId = admin.id;
    req.session.adminUsername = admin.username;
    res.json({ message: "تم تسجيل الدخول.", username: admin.username });
  });
});

app.post("/api/admin/logout", requireAdmin, (req, res) => {
  req.session.destroy(() => res.json({ message: "تم تسجيل الخروج." }));
});

app.get("/api/admin/me", requireAdmin, (req, res) => {
  res.json({ username: req.session.adminUsername });
});

app.get("/api/admin/requests", requireAdmin, (req, res) => {
  const rows = db.prepare("SELECT * FROM requests ORDER BY id DESC LIMIT 500").all();
  res.json(rows);
});

app.patch("/api/admin/requests/:id", requireAdmin, (req, res) => {
  const id = Number(req.params.id);
  const status = clean(req.body.status, 20);
  const allowed = ["new", "reviewing", "in_progress", "completed", "rejected"];
  if (!Number.isInteger(id) || !allowed.includes(status)) return res.status(400).json({ error: "بيانات غير صالحة." });
  const result = db.prepare("UPDATE requests SET status = ? WHERE id = ?").run(status, id);
  if (!result.changes) return res.status(404).json({ error: "الطلب غير موجود." });
  res.json({ message: "تم تحديث حالة الطلب." });
});

app.delete("/api/admin/requests/:id", requireAdmin, (req, res) => {
  const id = Number(req.params.id);
  if (!Number.isInteger(id)) return res.status(400).json({ error: "رقم الطلب غير صالح." });
  const result = db.prepare("DELETE FROM requests WHERE id = ?").run(id);
  if (!result.changes) return res.status(404).json({ error: "الطلب غير موجود." });
  res.json({ message: "تم حذف الطلب." });
});

app.get("/api/admin/stats", requireAdmin, (req, res) => {
  const total = db.prepare("SELECT COUNT(*) AS n FROM requests").get().n;
  const fresh = db.prepare("SELECT COUNT(*) AS n FROM requests WHERE status = 'new'").get().n;
  const active = db.prepare("SELECT COUNT(*) AS n FROM requests WHERE status IN ('reviewing','in_progress')").get().n;
  const completed = db.prepare("SELECT COUNT(*) AS n FROM requests WHERE status = 'completed'").get().n;
  res.json({ total, fresh, active, completed });
});

app.use(express.static(path.join(__dirname, "public"), { extensions: ["html"] }));
app.get("*", (req, res) => res.sendFile(path.join(__dirname, "public", "index.html")));

app.listen(PORT, () => console.log(`NOVYX website running at http://localhost:${PORT}`));

    button.disabled = false;

  }
});

const loginPanel = document.getElementById("loginPanel");
const dashboard = document.getElementById("dashboard");
const loginForm = document.getElementById("loginForm");
const loginMessage = document.getElementById("loginMessage");
const requestsList = document.getElementById("requestsList");
const statusLabels = { new: "جديد", reviewing: "قيد المراجعة", in_progress: "قيد التنفيذ", completed: "مكتمل", rejected: "مرفوض" };

async function api(url, options = {}) {
  const response = await fetch(url, { ...options, headers: { "Content-Type": "application/json", ...(options.headers || {}) } });
  const data = await response.json().catch(() => ({}));
  if (!response.ok) throw new Error(data.error || "حدث خطأ في الاتصال.");
  return data;
}
function showDashboard() { loginPanel.hidden = true; dashboard.hidden = false; loadDashboard(); }
async function checkSession() {
  try { await api("/api/admin/me"); showDashboard(); } catch { loginPanel.hidden = false; dashboard.hidden = true; }
}
loginForm.addEventListener("submit", async e => {
  e.preventDefault();
  const button = loginForm.querySelector("button");
  button.disabled = true; loginMessage.textContent = "";
  try {
    const payload = Object.fromEntries(new FormData(loginForm).entries());
    await api("/api/admin/login", { method: "POST", body: JSON.stringify(payload) });
    loginForm.reset(); showDashboard();
  } catch (err) { loginMessage.textContent = err.message; loginMessage.style.color = "#ff91ad"; }
  finally { button.disabled = false; }
});
document.getElementById("logoutBtn").addEventListener("click", async () => {
  try { await api("/api/admin/logout", { method: "POST" }); } finally { loginPanel.hidden = false; dashboard.hidden = true; }
});
document.getElementById("refreshBtn").addEventListener("click", loadDashboard);

async function loadDashboard() {
  try {
    const [stats, requests] = await Promise.all([api("/api/admin/stats"), api("/api/admin/requests")]);
    document.getElementById("statTotal").textContent = stats.total;
    document.getElementById("statFresh").textContent = stats.fresh;
    document.getElementById("statActive").textContent = stats.active;
    document.getElementById("statCompleted").textContent = stats.completed;
    renderRequests(requests);
  } catch (err) { requestsList.textContent = err.message; }
}
function make(tag, className, text) {
  const el = document.createElement(tag);
  if (className) el.className = className;
  if (text !== undefined) el.textContent = text;
  return el;
}
function renderRequests(requests) {
  requestsList.replaceChildren();
  if (!requests.length) { requestsList.append(make("p", "muted", "لا توجد طلبات حتى الآن.")); return; }
  requests.forEach(item => {
    const card = make("article", "request-card");
    const top = make("div", "request-top");
    const titleWrap = make("div");
    titleWrap.append(make("h3", "", `طلب #${item.id} — ${item.name}`));
    titleWrap.append(make("div", "request-meta", `${item.email} • ${item.phone} • ${item.service}`));
    top.append(titleWrap, make("strong", "", statusLabels[item.status] || item.status));
    card.append(top);
    card.append(make("p", "", item.details));
    if (item.budget) card.append(make("div", "request-meta", `الميزانية: ${item.budget}`));
    card.append(make("div", "request-meta", `تاريخ الاستلام: ${item.created_at}`));
    const actions = make("div", "request-actions");
    const select = document.createElement("select");
    Object.entries(statusLabels).forEach(([value, label]) => {
      const option = document.createElement("option"); option.value = value; option.textContent = label; option.selected = value === item.status; select.append(option);
    });
    select.addEventListener("change", async () => {
      try { await api(`/api/admin/requests/${item.id}`, { method: "PATCH", body: JSON.stringify({ status: select.value }) }); await loadDashboard(); }
      catch (err) { alert(err.message); }
    });
    const del = make("button", "small-btn danger", "حذف الطلب");
    del.addEventListener("click", async () => {
      if (!confirm(`هل تريد حذف الطلب رقم ${item.id}؟`)) return;
      try { await api(`/api/admin/requests/${item.id}`, { method: "DELETE" }); await loadDashboard(); }
      catch (err) { alert(err.message); }
    });
    actions.append(select, del); card.append(actions); requestsList.append(card);
  });
}
checkSession();

# NOVYX — Company Website

A responsive Arabic RTL company website with:
- Public services and company profile pages.
- Project request form saved to SQLite.
- Separate admin login and dashboard.
- Admin request status updates and deletion.
- Password hashing, secure session cookies, rate limits, and security headers.

## Requirements
- Node.js 20 or newer
- npm

## Run locally
1. Extract the ZIP.
2. Open the project folder in a terminal.
3. Install dependencies:
   `npm install`
4. Copy `.env.example` to `.env`.
5. Change `SESSION_SECRET` to a long random string (32+ characters).
6. Change `ADMIN_USERNAME` and `ADMIN_PASSWORD`. Use a unique password of at least 12 characters.
7. Start:
   `npm start`
8. Open `http://localhost:3000`.
9. Admin dashboard: `http://localhost:3000/admin.html`

The first startup creates the admin account if it does not already exist. The SQLite database is created in `data/novyx.sqlite`.

## Important deployment notes
- Deploy on a Node.js host that supports persistent disk storage.
- Set the environment variables in the host dashboard; do not upload `.env`.
- Set `NODE_ENV=production` and use HTTPS in production.
- Keep the `data/` directory on persistent storage and back it up.
- If you change `ADMIN_PASSWORD` after the admin record is created, the existing database password hash will not automatically change. For a password reset, back up the database, delete the admin row with a secure DB tool, and restart with the new environment values, or implement a password-change workflow.
- Do not use ephemeral/serverless hosting without persistent storage for this SQLite setup.
- The site currently stores customer requests in the database; it does not yet send email, WhatsApp messages, accept payments, or provide customer accounts.
